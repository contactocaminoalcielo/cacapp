# Instalado como /etc/cron.d/orbit-backend (root). El cron corre en UTC;
# Bogotá = UTC-5 sin DST. Los jobs se disparan a diario y cada uno decide
# si aplica según config_operativa (dias_planificacion).
# 06:00 Bogotá = 11:00 UTC → propuesta de lote (solo actúa lun/mié/vie)
# 06:30 Bogotá = 11:30 UTC → motor de alertas (todos los días)

# orbit-backend — jobs Tenjo (06:00/06:30 Bogota = 11:00/11:30 UTC)
0 11 * * * root curl -s -X POST -H "x-job-token: $(grep '^JOB_TOKEN=' /opt/orbit-backend/.env | cut -d= -f2-)" http://127.0.0.1:8787/jobs/generar-propuesta >> /var/log/orbit-backend-cron.log 2>&1
30 11 * * * root curl -s -X POST -H "x-job-token: $(grep '^JOB_TOKEN=' /opt/orbit-backend/.env | cut -d= -f2-)" http://127.0.0.1:8787/jobs/alertas >> /var/log/orbit-backend-cron.log 2>&1

# Reportes Grupales (Fase 5) — sincroniza reportes, marca vencidos y genera alertas.
# 06:45 Bogota = 11:45 UTC (todos los dias). Idempotente.
45 11 * * * root curl -s -X POST -H "x-job-token: $(grep '^JOB_TOKEN=' /opt/orbit-backend/.env | cut -d= -f2-)" http://127.0.0.1:8787/jobs/grupales >> /var/log/orbit-backend-cron.log 2>&1

# Solicitud de imagenes — prepara los contactos del dia (NO envia; una persona valida).
# 10:00 Bogota = 15:00 UTC (todos los dias). Idempotente; recupera vencidos no procesados.
0 15 * * * root curl -s -X POST -H "x-job-token: $(grep '^JOB_TOKEN=' /opt/orbit-backend/.env | cut -d= -f2-)" http://127.0.0.1:8787/jobs/contactos-imagenes >> /var/log/orbit-backend-cron.log 2>&1

# Seguimiento de imagenes — 2o contacto (+3 dias habiles) y 3er contacto (+15) desde
# el envio del 1o; cierra SIN_RESPUESTA + alerta tras el 3o. SOLO dias habiles (L-V):
# el cron no conoce los festivos de Colombia, pero el job SI (fn_sumar_dias_habiles);
# en festivo simplemente no habra nada vencido. 10:30 Bogota = 15:30 UTC.
# Idempotente: el indice unico (solicitud_id, numero) impide duplicar un contacto.
30 15 * * 1-5 root curl -s -X POST -H "x-job-token: $(grep '^JOB_TOKEN=' /opt/orbit-backend/.env | cut -d= -f2-)" http://127.0.0.1:8787/jobs/seguimiento-imagenes >> /var/log/orbit-backend-cron.log 2>&1
